async function fetchFromGoogle(text, dir, signal) { const sl = dir === "ar-en" ? "ar" : "en"; const tl = dir === "ar-en" ? "en" : "ar"; const params = `client=gtx&sl=${sl}&tl=${tl}&dt=t&dt=bd&q=${encodeURIComponent(text)}`; const url = `https://translate.googleapis.com/translate_a/single?${params}`; // 1) المحاولة المباشرة (سريعة إذا سمح المتصفح) try { const r = await fetch(url, { signal }); if (r.ok) return parseGoogleResponse(await r.json(), text); } catch (e) { if (e.name === "AbortError") throw e; } // 2) المرور عبر وسيط CORS عند الحظر const proxyUrl = `https://api.allorigins.win/raw?url=${encodeURIComponent(url)}`; try { const r2 = await fetch(proxyUrl, { signal }); if (r2.ok) return parseGoogleResponse(await r2.json(), text); } catch (e) { if (e.name === "AbortError") throw e; } throw new Error("net"); } function parseGoogleResponse(j, text) { if (!j || !j[0]) throw new Error("bad"); const translation = j[0].map(x => x && x[0] ? x[0] : "").join("").trim(); let dictData = null; if (!/\s/.test(text.trim()) && j[1]) { dictData = { word: text.trim(), meanings: [] }; for (const entry of j[1]) { if (!entry || !entry[0]) continue; const posAr = POS_MAP[entry[0].toLowerCase()] || entry[0]; const defs = (entry[1] || []).slice(0, 2).map((defEn, i) => ({ def: entry[2] && entry[2][i] ? entry[2][i] : defEn })); if (defs.length) dictData.meanings.push({ pos: posAr, defs }); } } return { translation, dictData }; }
